/*
	counting semaphore
	configUSE_PREEMPTION	1
	configUSE_COUNTING_SEMAPHORES	1

	task 1, task 2 give semaphore
	task 3 takes semaphore
*/



// INCLUDE HEADER FILES 
#include <stdio.h>
#include <conio.h>
#include "FreeRTOS.h"
#include "FreeRTOSConfig.h"
#include "task.h"
#include "semphr.h"

//FUNCTIONS
void func_1(void);
void func_2(void);
void func_3(void);

SemaphoreHandle_t Sem_Handle;
TaskHandle_t tsk1;
TaskHandle_t tsk2;
TaskHandle_t tsk3;

//MAIN FUNCTION
int main(void) {
	Sem_Handle = xSemaphoreCreateCounting(2, 0); //maxSEM = 2 ; initSEM = 0
	// The maximum count value that can be reached is 2
	// The init count value assigned to the semaphore is 0
	xTaskCreate(func_1, "TASK_1", configMINIMAL_STACK_SIZE, NULL, 3, &tsk1);
	xTaskCreate(func_2, "TASK_2", configMINIMAL_STACK_SIZE, NULL, 3, &tsk2);
	xTaskCreate(func_3, "TASK_3", configMINIMAL_STACK_SIZE, NULL, 4, &tsk3);
	vTaskStartScheduler();
	for (;;);
}

//TASKS 
void func_1(void) {
	while (1) {
		printf("%s give signal\n", pcTaskGetName(tsk1));
		xSemaphoreGive(Sem_Handle); //SEM++
		vTaskDelay(pdMS_TO_TICKS(1000));
	}
}

void func_2(void) {
	while (1) {
		printf("%s give signal\n", pcTaskGetName(tsk2));
		xSemaphoreGive(Sem_Handle); //SEM++		
		vTaskDelay(pdMS_TO_TICKS(1000));
	}
}

void func_3(void) {
	while (1) {
		xSemaphoreTake(Sem_Handle, portMAX_DELAY);
		printf("%s take SEM\n", pcTaskGetName(tsk3));
		xSemaphoreTake(Sem_Handle, portMAX_DELAY);
		printf("%s synchronized\n\n", pcTaskGetName(tsk3));
	}
}
