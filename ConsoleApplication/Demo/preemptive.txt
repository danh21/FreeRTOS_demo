/*
	2 tasks:
		task 1 lower priority
		task 2 higher priority

	if configUSE_PREEMPTION == 1 -> task 2 takes CPU while task 1 is running
*/



#include <stdio.h>  
#include <conio.h>
#include "FreeRTOS.h"
#include "FreeRTOSConfig.h"
#include "task.h"
#include "timers.h"
#include "queue.h"



void func_1(void);
void func_2(void);

TaskHandle_t tsk1; 
TaskHandle_t tsk2;



int main(void) {
	xTaskCreate(func_1, "TASK_1", configMINIMAL_STACK_SIZE, NULL, 2, tsk1);
	xTaskCreate(func_2, "TASK_2", configMINIMAL_STACK_SIZE, NULL, 3, tsk2);
	vTaskStartScheduler(); 
	for (;;);
}




void func_1(void) {
	int a, b;
	while (1) {
		printf("\n %s IS RUNNING \n", pcTaskGetName(tsk1)); 
		printf("Input a,b: ");
		scanf_s("%d%d", &a, &b);
		printf("sum = a + b = %d\n", a + b);
		vTaskDelay(pdMS_TO_TICKS(2000));  
	}
}

void func_2(void) {
	int stuff;
	vTaskDelay(pdMS_TO_TICKS(5000)); // sleep 5s
	while (1)
	{
		printf("\n %s IS RUNNING---------------- \n", pcTaskGetName(tsk2));
		stuff = INT_MAX;
		while (--stuff > 0);
	}
}
